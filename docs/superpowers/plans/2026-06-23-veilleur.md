# veilleur Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a daily autonomous tech-watch curator that runs matcha to fetch feeds, then scores/summarizes/links the items with an LLM and writes a Zettelkasten note into the Obsidian vault's `00_Inbox/`.

**Architecture:** `matcha` (pinned Go binary, treated as a black box) fetches + dedupes feeds and writes a raw markdown digest into a work dir. A Python package `veilleur` parses that markdown, triages items against an interests profile (1 batched LLM call), enriches the survivors (fetch article text → LLM summary + connections to existing permanent notes), and renders a daily note. matcha stays unpatched so upgrades are a version bump.

**Tech Stack:** Python 3.12+, `uv` (env + deps), `pytest` + `pytest-mock`, `anthropic` SDK, `httpx`, `trafilatura`. matcha installed via `go install github.com/piqoni/matcha@<version>`.

## Global Constraints

- Python `>=3.12` (uses stdlib `tomllib`).
- Content language of the produced note is **French**; module/code/identifier names in English.
- matcha is **never patched**. It is consumed as a pinned binary; the only coupling point is its markdown output, isolated in `digest_parser.py`.
- matcha must run with LLM summary, reading time, images, and instapaper **disabled** (config in `matcha.yaml.example`), so digest lines are bare links.
- Popularity signal is a **boolean `hot`** (matcha prefixes `🔥` on Hacker News items with ≥100 comments, `💬` otherwise). No numeric points exist in matcha output. Non-HN items have no signal (`hot=False`).
- Anthropic API key is read from the `ANTHROPIC_API_KEY` environment variable — never stored in config files.
- Default model: `claude-sonnet-4-6`. `triage_threshold = 6`, `max_items = 10`.
- Output note: `00_Inbox/YYYY-MM-DD-veille.md`, frontmatter `type: veille`, `status: ⏳`, `date`, `tags: [veille]`. Idempotent: skip if it exists unless `--force`.
- Fail-safe: if matcha fails, abort and write nothing. Per-item failures are isolated and flagged in the note, never silently dropped.
- All tests run with `uv run pytest`. No test may hit the network or call the real Anthropic API (mock the client / monkeypatch `llm.call_json` and `time.sleep`).

import os
import tomllib
from dataclasses import dataclass
from pathlib import Path


@dataclass
class Config:
    vault_path: Path
    permanent_notes_dir: Path
    inbox_dir: Path
    interests_path: Path
    work_dir: Path

        triage_threshold=int(data["triage_threshold"]),

`src/veilleur/digest_parser.py`:

```python
import re

from .models import Item

_LINK = re.compile(r"\[([^\]]*)\]\(([^)]+)\)")
_HEADER = re.compile(r"^#{2,3}\s+(.*)$")
_IMG = re.compile(r"<img[^>]*>")
_ICONS = {"💬", "🔥"}


def _clean_source(raw: str) -> str:
    raw = _IMG.sub("", raw)
    return raw.replace("🍵", "").strip()


def parse_digest(markdown: str) -> list[Item]:
    items: list[Item] = []
    source = "Inconnu"
    for raw in markdown.splitlines():
        line = raw.strip()
        if not line:
            continue
        header = _HEADER.match(line)
        if header:
            source = _clean_source(header.group(1))
            continue
        links = _LINK.findall(line)
        if not links:
            continue
        hot = False
        if len(links) >= 2 and links[0][0].strip() in _ICONS:
            hot = links[0][0].strip() == "🔥"
            title, url = links[1][0].strip(), links[1][1].strip()
        else:
            title, url = links[0][0].strip(), links[0][1].strip()
        if not title or not url:
            continue
        items.append(Item(title=title, url=url, source=source, hot=hot))
    return items
```

`src/veilleur/vault_index.py`:

```python
import hashlib
import json
import re
from dataclasses import asdict, dataclass
from pathlib import Path

_CONCEPT = re.compile(r"##\s*💡[^\n]*\n(.*?)(?=\n##|\Z)", re.DOTALL)
_FRONTMATTER = re.compile(r"^---\n(.*?)\n---", re.DOTALL)


@dataclass
class VaultNote:
    title: str
    concept: str
    tags: list[str]


def _parse_concept(text: str) -> str:
    m = _CONCEPT.search(text)
    if not m:
        return ""
    body = m.group(1).strip()
    # Strip leading blockquote markers and whitespace from each line.
    lines = [ln.lstrip("> ").strip() for ln in body.splitlines()]
    return " ".join(ln for ln in lines if ln)


def _parse_tags(text: str) -> list[str]:
    fm = _FRONTMATTER.search(text)
    if not fm:
        return []
    tags: list[str] = []
    in_tags = False
    for line in fm.group(1).splitlines():
        if re.match(r"^tags:\s*$", line):
            in_tags = True
            continue
        if in_tags:
            m = re.match(r"^\s*-\s*(.+?)\s*$", line)
            if m:
                tags.append(m.group(1))
            elif line.strip() and not line.startswith(" "):
                break
    return tags


def parse_note(path: Path) -> VaultNote:
    text = path.read_text(encoding="utf-8")
    return VaultNote(
        title=path.stem,
        concept=_parse_concept(text),
        tags=_parse_tags(text),
    )


def build_index(permanent_dir: Path) -> list[VaultNote]:
    return [parse_note(p) for p in sorted(permanent_dir.glob("*.md"))]


def _signature(permanent_dir: Path) -> str:
    parts = sorted(
        (p.name, p.stat().st_mtime_ns) for p in permanent_dir.glob("*.md")
    )
    return hashlib.sha256(repr(parts).encode()).hexdigest()


def build_index_cached(permanent_dir: Path, cache_path: Path) -> list[VaultNote]:
    sig = _signature(permanent_dir)
    if cache_path.exists():
        cached = json.loads(cache_path.read_text(encoding="utf-8"))
        if cached.get("sig") == sig:
            return [VaultNote(**n) for n in cached["notes"]]
    notes = build_index(permanent_dir)
    cache_path.parent.mkdir(parents=True, exist_ok=True)
    cache_path.write_text(
        json.dumps({"sig": sig, "notes": [asdict(n) for n in notes]}),
        encoding="utf-8",
    )
    return notes


def format_index(notes: list[VaultNote]) -> str:
    lines = []
    for n in notes:
        tagstr = f" [tags: {', '.join(n.tags)}]" if n.tags else ""
        lines.append(f"- [[{n.title}]]: {n.concept}{tagstr}")
    return "\n".join(lines)
```

import httpx
import trafilatura

_HEADERS = {"User-Agent": "veilleur/0.1 (+https://github.com/alexandre)"}


def fetch_text(url: str, *, timeout: float = 20.0, max_chars: int = 8000) -> str | None:
    try:
        resp = httpx.get(url, timeout=timeout, follow_redirects=True, headers=_HEADERS)
        resp.raise_for_status()
    except Exception:
        return None
    extracted = trafilatura.extract(resp.text)
    if not extracted:
        return None
    return extracted[:max_chars]
```

```python
import json
import time

import anthropic


class LLMError(Exception):
    pass


def make_client(api_key: str) -> anthropic.Anthropic:
    return anthropic.Anthropic(api_key=api_key)


def _extract_json(text: str) -> str:
    text = text.strip()
    if text.startswith("```"):
        inner = text.split("```")[1]
        if inner.startswith("json"):
            inner = inner[len("json"):]
        return inner.strip()
    return text


def call_json(client, model, system, user, *, max_tokens=2048, retries=3, sleep=time.sleep):
    last = None
    for attempt in range(retries):
        try:
            msg = client.messages.create(
                model=model,
                max_tokens=max_tokens,
                system=system,
                messages=[{"role": "user", "content": user}],
            )
            return json.loads(_extract_json(msg.content[0].text))
        except Exception as exc:  # network, parse, API errors all retry
            last = exc
            if attempt < retries - 1:
                sleep(2 ** attempt)
    raise LLMError(f"LLM call failed after {retries} attempts: {last}")
```

from .models import Item

_SYSTEM = (
    "Tu es l'assistant de veille techno d'un ingénieur qui vise l'expertise. "
    "Tu notes des articles selon leur valeur d'apprentissage pour LUI, en "
    "fonction de son profil d'intérêts. Réponds UNIQUEMENT en JSON."
)

_INSTRUCTIONS = (
    "Pour CHAQUE article, donne un score de 0 à 10 (pertinence vis-à-vis du "
    "profil + valeur d'apprentissage; un article marqué [HOT] est populaire, "
    "pondère légèrement à la hausse mais la popularité ne prime pas sur la "
    "pertinence). Donne une raison courte en français et 1 à 3 tags en "
    'kebab-case. Format STRICT: {"scores": [{"index": <int>, "score": <int>, '
    '"reason": "<fr>", "tags": ["<tag>"]}]}'
)


def triage(items, interests, client, model, threshold, max_items):
    if not items:
        return []
    listing = "\n".join(
        f"{i}. [{'HOT' if it.hot else 'normal'}] {it.title} ({it.source})"
        for i, it in enumerate(items)
    )
    user = f"PROFIL D'INTÉRÊTS:\n{interests}\n\nARTICLES:\n{listing}\n\n{_INSTRUCTIONS}"
    data = llm.call_json(client, model, _SYSTEM, user, max_tokens=2048)
    by_index = {int(d["index"]): d for d in data.get("scores", [])}
    for i, it in enumerate(items):
        d = by_index.get(i)
        if d is None:
            continue
        it.score = int(d["score"])
        it.score_reason = d.get("reason", "")
        it.tags = list(d.get("tags", []))
    kept = [it for it in items if it.score is not None and it.score >= threshold]
    kept.sort(key=lambda it: it.score, reverse=True)
    return kept[:max_items]
```

_SYSTEM = (
    "Tu es l'assistant de veille d'un ingénieur. Pour un article donné, tu "
    "produis un résumé en français, expliques pourquoi il vaut son temps au "
    "regard de sa montée en expertise, et proposes des connexions UNIQUEMENT "

    content = fetch(item.url)
    body = content if content else "(contenu indisponible — base-toi sur le titre)"
    user = (
        f"ARTICLE: {item.title}\nSOURCE: {item.source}\nURL: {item.url}\n\n"
        f"CONTENU:\n{body}\n\n"

        Connection(note_title=c["note"], reason=c.get("reason", ""))

from .models import Item


def render_note(items: list[Item], date_str: str, n_analyzed: int) -> str:
    lines = [
        "---",
        "type: veille",
        "status: ⏳",
        f"date: {date_str}",
        "tags:",
        "  - veille",
        "---",
        "",
        f"# 📡 Veille — {date_str}",
        "",
        f"> {len(items)} articles retenus sur {n_analyzed} analysés.",
        "",
    ]
    for it in items:
        lines.append(f"## [{it.title}]({it.url})")
        hot = " · 🔥" if it.hot else ""
        score = f" · **Score :** {it.score}/10" if it.score is not None else ""
        lines.append(f"**Source :** {it.source}{hot}{score}")
        if it.tags:
            lines.append(f"**Tags :** {', '.join(it.tags)}")
        lines.append("")
        if it.enrich_failed:
            lines.append("_Résumé indisponible (enrichissement échoué)._")
            lines.append("")
        else:
            if it.summary:
                lines.append(it.summary)
                lines.append("")
            if it.why:
                lines.append(f"**Pourquoi pour toi :** {it.why}")
                lines.append("")
        lines.append("### 🔗 Connexions")
        if it.connections:
            for c in it.connections:
                lines.append(f"- [[{c.note_title}]] — {c.reason}")
        else:
            lines.append("- [[ ]]")
        lines.append("")
        lines.append("---")
        lines.append("")
    return "\n".join(lines).rstrip() + "\n"
```

import subprocess
from datetime import date
from pathlib import Path


class MatchaError(Exception):
    pass


def run_matcha(cfg, today: str | None = None) -> Path:
    today = today or date.today().isoformat()
    cmd = [cfg.matcha_bin, "-c", str(cfg.matcha_config)]
    try:
        subprocess.run(
            cmd,
            cwd=str(cfg.work_dir),
            check=True,
            capture_output=True,
            text=True,
            timeout=300,
        )
    except (subprocess.CalledProcessError, FileNotFoundError,
            subprocess.TimeoutExpired) as exc:
        raise MatchaError(f"matcha run failed: {exc}") from exc
    out = cfg.work_dir / f"{cfg.matcha_file_prefix}{today}{cfg.matcha_file_suffix}.md"
    if not out.exists():
        raise MatchaError(f"matcha produced no output at {out}")
    return out
```

```python
import argparse
from datetime import date
from pathlib import Path

from . import (
    config,
    content_fetch,
    digest_parser,
    enrich,
    llm,
    matcha_runner,
    render,
    triage,
    vault_index,
)


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(prog="veilleur")
    parser.add_argument("--config", default="~/.config/veilleur/veilleur.toml")
    parser.add_argument("--force", action="store_true")
    parser.add_argument("--date", default=None)
    args = parser.parse_args(argv)

    cfg = config.load_config(Path(args.config).expanduser())
    today = args.date or date.today().isoformat()

    note_path = cfg.inbox_dir / f"{today}-veille.md"
    if note_path.exists() and not args.force:
        print(f"Note déjà présente: {note_path} (utilise --force pour réécrire)")
        return 0

    digest_path = matcha_runner.run_matcha(cfg, today=today)
    items = digest_parser.parse_digest(digest_path.read_text(encoding="utf-8"))
    n_analyzed = len(items)
    if not items:
        print("Aucun item dans le digest matcha.")
        return 0

    client = llm.make_client(cfg.anthropic_api_key)
    interests = cfg.interests_path.read_text(encoding="utf-8")
    notes = vault_index.build_index_cached(
        cfg.permanent_notes_dir, cfg.work_dir / "vault_index.json")
    index_str = vault_index.format_index(notes)

    kept = triage.triage(items, interests, client, cfg.model,
                         cfg.triage_threshold, cfg.max_items)
    for item in kept:
        enrich.enrich_item(item, index_str, client, cfg.model,
                           content_fetch.fetch_text)

    note = render.render_note(kept, today, n_analyzed)
    note_path.parent.mkdir(parents=True, exist_ok=True)
    note_path.write_text(note, encoding="utf-8")
    print(f"Écrit: {note_path} ({len(kept)} items retenus sur {n_analyzed})")
    return 0
```
